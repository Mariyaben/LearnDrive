cmake_minimum_required(VERSION 3.22)
project(LearnDrive LANGUAGES CXX)
if(WIN32 AND CMAKE_CXX_COMPILER MATCHES "zig-cxx")
  set(CMAKE_AR "${CMAKE_SOURCE_DIR}/tools/zig-ar.cmd")
  set(CMAKE_CXX_ARCHIVE_FINISH "")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
option(LEARNDRIVE_WITH_ORT "Build the native ONNX causal SLM decoder" OFF)
option(LEARNDRIVE_WITH_SPATIAL_ORT "Build the shared Supercombo ONNX inference runtime" OFF)
option(LEARNDRIVE_WITH_HF_TOKENIZER "Build the tokenizers-cpp tokenizer adapter and live SLM harness" OFF)
add_library(learndrive_core STATIC core/src/fusion.cpp core/src/decision.cpp core/src/vision.cpp core/src/c_api.cpp
  core/src/perception/camera_calibration.cpp core/src/perception/metric_estimator.cpp
  core/src/perception/supercombo.cpp
  core/src/SceneState.cpp core/src/ReflexEngine.cpp core/src/PriorityArbiter.cpp
  core/src/ContextualCognitionEngine.cpp core/src/DualBrainEngine.cpp core/src/dual_brain_c_api.cpp)
target_sources(learndrive_core PRIVATE core/src/PostDriveSummary.cpp)
target_sources(learndrive_core PRIVATE core/src/decision_benchmark.cpp)
target_sources(learndrive_core PRIVATE core/src/ContextualPolicyEngine.cpp)
target_sources(learndrive_core PRIVATE core/src/navigation/polyline_analyzer.cpp)
target_link_libraries(learndrive_core PUBLIC Threads::Threads)
set_target_properties(learndrive_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(learndrive_core PUBLIC core/include)
if(MSVC)
  target_compile_options(learndrive_core PRIVATE /W4)
else()
  target_compile_options(learndrive_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(LEARNDRIVE_WITH_SPATIAL_ORT)
  if(ANDROID)
    set(ONNXRUNTIME_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/android/ort-sdk/headers")
    set(ONNXRUNTIME_LIBRARY "${CMAKE_SOURCE_DIR}/android/ort-sdk/jni/${ANDROID_ABI}/libonnxruntime.so")
  else()
    find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h HINTS "${ONNXRUNTIME_ROOT}/include" REQUIRED)
    find_library(ONNXRUNTIME_LIBRARY NAMES onnxruntime HINTS "${ONNXRUNTIME_ROOT}/lib" REQUIRED)
  endif()
  target_sources(learndrive_core PRIVATE core/src/perception/supercombo_runtime.cpp)
  target_include_directories(learndrive_core PRIVATE "${ONNXRUNTIME_INCLUDE_DIR}")
  target_link_libraries(learndrive_core PUBLIC "${ONNXRUNTIME_LIBRARY}" ${CMAKE_DL_LIBS})
  target_compile_definitions(learndrive_core PUBLIC LEARNDRIVE_WITH_SPATIAL_ORT=1)
endif()
if(LEARNDRIVE_WITH_ORT)
  set(ONNXRUNTIME_ROOT "" CACHE PATH "ORT native SDK with include/ and lib/ for the target architecture")
  find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h HINTS "${ONNXRUNTIME_ROOT}/include" REQUIRED)
  find_library(ONNXRUNTIME_LIBRARY NAMES onnxruntime HINTS "${ONNXRUNTIME_ROOT}/lib" REQUIRED)
  target_sources(learndrive_core PRIVATE core/src/OnnxLanguageModel.cpp)
  target_include_directories(learndrive_core PRIVATE "${ONNXRUNTIME_INCLUDE_DIR}")
  target_link_libraries(learndrive_core PUBLIC "${ONNXRUNTIME_LIBRARY}")
  target_compile_definitions(learndrive_core PUBLIC LEARNDRIVE_WITH_ORT=1)
endif()
if(LEARNDRIVE_WITH_HF_TOKENIZER)
  if(NOT LEARNDRIVE_WITH_ORT)
    message(FATAL_ERROR "LEARNDRIVE_WITH_HF_TOKENIZER requires LEARNDRIVE_WITH_ORT")
  endif()
  set(TOKENIZERS_CPP_SOURCE_DIR "" CACHE PATH "Pinned tokenizers-cpp source checkout with submodules")
  if(NOT EXISTS "${TOKENIZERS_CPP_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Set TOKENIZERS_CPP_SOURCE_DIR to a tokenizers-cpp checkout")
  endif()
  add_subdirectory("${TOKENIZERS_CPP_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/tokenizers-cpp" EXCLUDE_FROM_ALL)
  target_sources(learndrive_core PRIVATE core/src/HuggingFaceTokenizer.cpp)
  target_compile_definitions(learndrive_core PUBLIC LEARNDRIVE_WITH_HF_TOKENIZER=1)
  target_link_libraries(learndrive_core PUBLIC tokenizers_cpp)
endif()
if(ANDROID)
  add_library(learndrive SHARED android/app/src/main/cpp/jni.cpp android/app/src/main/cpp/onnx_bridge.cpp)
  target_link_libraries(learndrive PRIVATE learndrive_core android log)
else()
  add_executable(learndrive_desk desktop/main.cpp)
  target_link_libraries(learndrive_desk PRIVATE learndrive_core Threads::Threads)
  enable_testing()
  add_executable(test_offline_geometry core/tests/test_offline_geometry.cpp)
  target_link_libraries(test_offline_geometry PRIVATE learndrive_core)
  add_test(NAME offline_geometry COMMAND test_offline_geometry)
  add_executable(guidance_pipeline_tests tests/guidance_pipeline_tests.cpp)
  target_link_libraries(guidance_pipeline_tests PRIVATE learndrive_core)
  add_test(NAME guidance_pipeline COMMAND guidance_pipeline_tests)
  add_executable(spatial_integration_tests tests/spatial_integration_tests.cpp)
  target_link_libraries(spatial_integration_tests PRIVATE learndrive_core)
  add_test(NAME spatial_integration COMMAND spatial_integration_tests)
  if(LEARNDRIVE_WITH_SPATIAL_ORT)
    add_library(learndrive_replay SHARED desktop/replay_bridge.cpp)
    target_link_libraries(learndrive_replay PRIVATE learndrive_core)
    add_executable(supercombo_smoke tests/supercombo_smoke.cpp)
    target_link_libraries(supercombo_smoke PRIVATE learndrive_core)
    add_test(NAME supercombo_real_model COMMAND supercombo_smoke
      "${CMAKE_SOURCE_DIR}/android/app/src/main/assets/supercombo.onnx" "${CMAKE_BINARY_DIR}/supercombo-smoke")
  endif()
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE learndrive_core Threads::Threads)
  add_test(NAME core_tests COMMAND core_tests)
  add_executable(dual_brain_tests tests/dual_brain_tests.cpp)
  target_link_libraries(dual_brain_tests PRIVATE learndrive_core)
  add_test(NAME dual_brain_tests COMMAND dual_brain_tests)
  add_executable(dual_brain_harness desktop/dual_brain_harness.cpp)
  target_link_libraries(dual_brain_harness PRIVATE learndrive_core)
  add_test(NAME dual_brain_scenario COMMAND dual_brain_harness)
  add_executable(test_reflex_suppression core/tests/test_reflex_suppression.cpp)
  target_link_libraries(test_reflex_suppression PRIVATE learndrive_core)
  add_test(NAME reflex_suppression COMMAND test_reflex_suppression)
  add_executable(test_priority_arbiter core/tests/test_priority_arbiter.cpp)
  target_link_libraries(test_priority_arbiter PRIVATE learndrive_core)
  add_test(NAME priority_arbiter COMMAND test_priority_arbiter)
  add_executable(test_perception_geometry core/tests/test_perception_geometry.cpp)
  target_link_libraries(test_perception_geometry PRIVATE learndrive_core)
  add_test(NAME perception_geometry COMMAND test_perception_geometry)
  if(LEARNDRIVE_WITH_ORT)
    add_executable(onnx_decoder_tests tests/onnx_decoder_tests.cpp)
    target_link_libraries(onnx_decoder_tests PRIVATE learndrive_core)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME create_decoder_fixture COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/create_decoder_fixture.py" "${CMAKE_BINARY_DIR}/decoder-fixture.onnx")
    set_tests_properties(create_decoder_fixture PROPERTIES FIXTURES_SETUP decoder_fixture)
    add_test(NAME onnx_decoder_tests COMMAND onnx_decoder_tests "${CMAKE_BINARY_DIR}/decoder-fixture.onnx")
    set_tests_properties(onnx_decoder_tests PROPERTIES FIXTURES_REQUIRED decoder_fixture)
  endif()
endif()
