cmake_minimum_required(VERSION 3.22)
project(LearnDrive LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(learndrive_core STATIC core/src/fusion.cpp core/src/decision.cpp core/src/vision.cpp core/src/c_api.cpp)
set_target_properties(learndrive_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(learndrive_core PUBLIC core/include)
if(MSVC)
  target_compile_options(learndrive_core PRIVATE /W4)
else()
  target_compile_options(learndrive_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(ANDROID)
  add_library(learndrive SHARED android/app/src/main/cpp/jni.cpp)
  target_link_libraries(learndrive PRIVATE learndrive_core android log)
else()
  find_package(Threads REQUIRED)
  add_executable(learndrive_desk desktop/main.cpp)
  target_link_libraries(learndrive_desk PRIVATE learndrive_core Threads::Threads)
  enable_testing()
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE learndrive_core Threads::Threads)
  add_test(NAME core_tests COMMAND core_tests)
endif()
